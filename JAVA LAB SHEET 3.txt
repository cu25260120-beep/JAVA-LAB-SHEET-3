Q1. Write a Java program to input and display 5 integers using a 1-D array.

Program:

import java.util.Scanner;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] a = new int[5];

        for (int i = 0; i < 5; i++)
            a[i] = sc.nextInt();

        for (int n : a)
            System.out.print(n + " ");
    }
}

Output:

Input: 10 20 30 40 50

Q13. Write a Java program to multiply two 3 × 3 matrices.

Program:

class Main {
    public static void main(String[] args) {
        int[][] a = {{1, 2, 3}, {4, 5, 6}, {7, 8, 9}};
        int[][] b = {{1, 0, 0}, {0, 1, 0}, {0, 0, 1}};
        int[][] c = new int[3][3];

        for (int i = 0; i < 3; i++)
            for (int j = 0; j < 3; j++)
                for (int k = 0; k < 3; k++)
                    c[i][j] += a[i][k] * b[k][j];

        for (int[] row : c) {
            for (int n : row)
                System.out.print(n + " ");
            System.out.println();
        }
    }
}

Output:

1 2 3
4 5 6

Q22. Count how many times each element appears in a 1-D array.

Program:

class Main {
    public static void main(String[] args) {
        int[] a = {1, 2, 2, 3, 1, 2};
        boolean[] visited = new boolean[a.length];

        for (int i = 0; i < a.length; i++) {
            if (visited[i]) continue;
            int count = 1;

            for (int j = i + 1; j < a.length; j++)
                if (a[i] == a[j]) {
                    count++;
                    visited[j] = true;
                }

            System.out.println(a[i] + " occurs " + count + " times");
        }
    }
}

Output:

1 occurs 2 times
2 occurs 3 times
3 occurs 1 times

Q24. Remove duplicate elements from a 1-D array without using collections.

Program:

class Main {
    public static void main(String[] args) {
        int[] a = {1, 2, 2, 3, 1, 4};
        int[] b = new int[a.length];
        int size = 0;

        for (int n : a) {
            boolean found = false;
            for (int i = 0; i < size; i++)
                if (b[i] == n) found = true;

            if (!found) b[size++] = n;
        }

        for (int i = 0; i < size; i++)
            System.out.print(b[i] + " ");
    }
}

Output:

1 2 3 4

Q30. Check if a given matrix is a sparse matrix (more zero elements than non-zero elements).

Program:

class Main {
    public static void main(String[] args) {
        int[][] a = {{0, 0, 3}, {0, 0, 0}, {4, 0, 0}};
        int zero = 0, nonZero = 0;

        for (int[] row : a)
            for (int n : row) {
                if (n == 0) zero++;
                else nonZero++;
            }

        if (zero > nonZero)
            System.out.println("Sparse Matrix");
        else
            System.out.println("Not a Sparse Matrix");
    }
}

Output:

Sparse Matrix
